  224: function resolveTiming(params: {
  225:   body: ActivityRecordBody;
  226:   durationMinutes: number | null;
  227:   requireDuration: boolean;
  228: }): ResolvedTiming | { error: string } {
  229:   const { body, durationMinutes, requireDuration } = params;
  230: 
  231:   const rawStartedAt = asString(body.startedAt) ?? asString(body.startTime);
  232:   const rawEndedAt = asString(body.endedAt) ?? asString(body.endTime);
  233: 
  234:   if (durationMinutes !== null && durationMinutes < 0) {
  235:     return { error: "durationMinutes must be greater than or equal to 0" };
  236:   }
  237: 
  238:   if (rawStartedAt && rawEndedAt) {
  239:     const startedDate = new Date(rawStartedAt);
  240:     const endedDate = new Date(rawEndedAt);
  241: 
  242:     if (
  243:       Number.isNaN(startedDate.getTime()) ||
  244:       Number.isNaN(endedDate.getTime())
  245:     ) {
  246:       return { error: "Invalid startedAt or endedAt" };
  247:     }
  248: 
  249:     if (endedDate.getTime() < startedDate.getTime()) {
  250:       return { error: "endedAt must be greater than or equal to startedAt" };
  251:     }
  252: 
  253:     const calculatedDurationMinutes = Math.round(
  254:       (endedDate.getTime() - startedDate.getTime()) / 60000
  255:     );
  256: 
  257:     return {
  258:       startedAt: startedDate.toISOString(),
  259:       endedAt: endedDate.toISOString(),
  260:       durationMinutes: calculatedDurationMinutes,
  261:     };
  262:   }
  263: 
  264:   if (rawStartedAt) {
  265:     const startedDate = new Date(rawStartedAt);
  266: 
  267:     if (Number.isNaN(startedDate.getTime())) {
  268:       return { error: "Invalid startedAt" };
  269:     }
  270: 
  271:     if (durationMinutes === null) {
  272:       if (requireDuration) {
  273:         return { error: "durationMinutes is required for this activity" };
  274:       }
  275: 
  276:       return {
  277:         startedAt: startedDate.toISOString(),
  278:         endedAt: null,
  279:         durationMinutes: null,
  280:       };
  281:     }
  282: 
  283:     const endedDate = new Date(startedDate.getTime() + durationMinutes * 60000);
  284: 
  285:     return {
  286:       startedAt: startedDate.toISOString(),
  287:       endedAt: endedDate.toISOString(),
  288:       durationMinutes,
  289:     };
  290:   }
  291: 
  292:   if (rawEndedAt) {
  293:     const endedDate = new Date(rawEndedAt);
  294: 
  295:     if (Number.isNaN(endedDate.getTime())) {
  296:       return { error: "Invalid endedAt" };
  297:     }
  298: 
  299:     if (durationMinutes === null) {
  300:       if (requireDuration) {
  301:         return { error: "durationMinutes is required for this activity" };
  302:       }
  303: 
  304:       return {
  305:         startedAt: null,
  306:         endedAt: endedDate.toISOString(),
  307:         durationMinutes: null,
  308:       };
  309:     }
  310: 
  311:     const startedDate = new Date(endedDate.getTime() - durationMinutes * 60000);
  312: 
  313:     return {
  314:       startedAt: startedDate.toISOString(),
  315:       endedAt: endedDate.toISOString(),
  316:       durationMinutes,
  317:     };
  318:   }
  319: 
  320:   if (durationMinutes === null) {
  321:     if (requireDuration) {
  322:       return { error: "durationMinutes is required for this activity" };
  323:     }
  324: 
  325:     return {
  326:       startedAt: null,
  327:       endedAt: null,
  328:       durationMinutes: null,
  329:     };
  330:   }
  331: 
  332:   const endedDate = new Date();
  333:   const startedDate = new Date(endedDate.getTime() - durationMinutes * 60000);
  334: 
  335:   return {
  336:     startedAt: startedDate.toISOString(),
  337:     endedAt: endedDate.toISOString(),
  338:     durationMinutes,
  339:   };
  340: }
