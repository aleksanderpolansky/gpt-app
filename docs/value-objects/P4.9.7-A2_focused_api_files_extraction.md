  348:   column?: string;
  349:   eventIds: string[];
  350:   limit: number;
  351:   orderBy?: string;
  352:   ascending?: boolean;
  353: }) {
  354:   const { table, column, eventIds, limit, orderBy, ascending } = params;
  355: 
  356:   if (eventIds.length === 0) {
  357:     return [];
  358:   }
  359: 
  360:   let query = supabase
  361:     .from(table)
  362:     .select("*")
  363:     .in(column ?? "event_id", eventIds)
  364:     .limit(limit);
  365: 
  366:   if (orderBy) {
  367:     query = query.order(orderBy, { ascending: ascending ?? false });
  368:   }
  369: 
  370:   const { data, error } = await query;
  371: 
  372:   if (error) {
  373:     throw new Error(`${table}: ${error.message}`);
  374:   }
  375: 
  376:   return (data as GenericRow[] | null) ?? [];

  881: function buildImpactSummary(rows: GenericRow[]) {
  882:   const groups = new Map<
  883:     string,
  884:     {
  885:       targetType: string | null;
  886:       targetKey: string | null;
  887:       metric: string | null;
  888:       direction: string | null;
  889:       unit: string | null;
  890:       count: number;
  891:       numericSum: number | null;
  892:       eventIds: string[];
  893:       sampleImpactIds: string[];
  894:       latestCreatedAt: string | null;
  895:     }
  896:   >();
  897: 
  898:   for (const row of rows) {
  899:     const targetType = getFirstStringField(row, ["impact_target_type", "target_type"]);
  900:     const targetKey = getFirstStringField(row, ["impact_target_key", "target_key"]);
  901:     const metric = getFirstStringField(row, ["impact_metric", "metric"]);
  902:     const direction = getFirstStringField(row, ["impact_direction", "direction"]);
  903:     const unit = getFirstStringField(row, ["impact_unit", "unit"]);
  904:     const key = JSON.stringify([targetType, targetKey, metric, direction, unit]);
  905:     const numericValue = getFirstNumberField(row, ["impact_value_numeric", "value_numeric"]);
  906:     const eventId = getStringField(row, "event_id");
  907:     const impactId = getId(row);
  908:     const createdAt = getStringField(row, "created_at");
  909: 
  910:     const existing =
  911:       groups.get(key) ??
  912:       {
  913:         targetType,
  914:         targetKey,
  915:         metric,
  916:         direction,
  917:         unit,
  918:         count: 0,
  919:         numericSum: null,
  920:         eventIds: [],
  921:         sampleImpactIds: [],
  922:         latestCreatedAt: null,
  923:       };
  924: 
  925:     existing.count += 1;
  926: 
  927:     if (numericValue !== null) {
  928:       existing.numericSum = (existing.numericSum ?? 0) + numericValue;
  929:     }
  930: 
  931:     if (eventId && !existing.eventIds.includes(eventId)) {
  932:       existing.eventIds.push(eventId);
  933:     }
  934: 
  935:     if (impactId && existing.sampleImpactIds.length < 10) {
  936:       existing.sampleImpactIds.push(impactId);
  937:     }
  938: 
  939:     if (
  940:       createdAt &&
  941:       (!existing.latestCreatedAt ||
  942:         Date.parse(createdAt) > Date.parse(existing.latestCreatedAt))
  943:     ) {
  944:       existing.latestCreatedAt = createdAt;
  945:     }
  946: 
  947:     groups.set(key, existing);
  948:   }
  949: 
  950:   return Array.from(groups.values()).sort((left, right) => {
  951:     const leftTime = left.latestCreatedAt ? Date.parse(left.latestCreatedAt) : 0;
  952:     const rightTime = right.latestCreatedAt ? Date.parse(right.latestCreatedAt) : 0;
  953: 
  954:     return rightTime - leftTime;
  955:   });
  956: }
