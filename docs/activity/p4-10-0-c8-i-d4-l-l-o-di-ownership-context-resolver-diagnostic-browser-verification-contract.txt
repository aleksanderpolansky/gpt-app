BROWSER VERIFICATION SCRIPT FOR FUTURE O-DJ EXECUTION ONLY

IMPORTANT:
This script is documentation for the next runtime/browser verification gate.
O-DI does not execute this script.

Run in browser DevTools Console at:
http://localhost:3000

The user must be authenticated for authenticated cases.

(async () => {
  const endpoint = "/api/activity/debug/ownership-context-resolution-read-only-test";
  const requiredHeaderName = "x-controlled-intake-ownership-context-resolution-read-only-test";
  const requiredHeaderValue = "enabled";

  const results = [];

  async function runCase(label, url, options = {}) {
    const response = await fetch(url, {
      method: "GET",
      credentials: "include",
      cache: "no-store",
      ...options,
    });

    let body = null;

    try {
      body = await response.json();
    } catch (error) {
      body = {
        jsonParseError: error instanceof Error ? error.message : String(error),
      };
    }

    results.push({
      label,
      httpStatus: response.status,
      ok: response.ok,
      body,
    });

    return {
      response,
      body,
    };
  }

  await runCase("missing_header", endpoint);

  await runCase("wrong_header", endpoint, {
    headers: {
      [requiredHeaderName]: "wrong",
    },
  });

  await runCase("valid_header_no_context_hints", endpoint, {
    headers: {
      [requiredHeaderName]: requiredHeaderValue,
    },
  });

  await runCase("valid_header_invalid_space_id", `${endpoint}?spaceId=not-a-uuid`, {
    headers: {
      [requiredHeaderName]: requiredHeaderValue,
    },
  });

  await runCase(
    "valid_header_ambiguous_space_and_organization",
    `${endpoint}?spaceId=00000000-0000-4000-8000-000000000001&organizationId=00000000-0000-4000-8000-000000000002`,
    {
      headers: {
        [requiredHeaderName]: requiredHeaderValue,
      },
    },
  );

  console.log("O_DI_BROWSER_VERIFICATION_RESULTS");
  console.log(JSON.stringify(results, null, 2));
})();
