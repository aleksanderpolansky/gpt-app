IMPORTANT:

Audit target:
Find exact public schema tables, columns, constraints, indexes, RLS status and policies related to:
- actors
- organizations
- spaces
- memberships
- actor/organization/space/user/app_user links
- status fields
- role fields

Recommended metadata query group A: table candidates
SELECT table_schema, table_name
FROM information_schema.tables
WHERE table_schema IN ('public')
  AND (
    table_name ILIKE '%actor%'
    OR table_name ILIKE '%organization%'
    OR table_name ILIKE '%space%'
    OR table_name ILIKE '%member%'
    OR table_name ILIKE '%role%'
    OR table_name ILIKE '%user%'
    OR table_name ILIKE '%app_user%'
  )
ORDER BY table_schema, table_name;

Recommended metadata query group B: column candidates
SELECT table_schema, table_name, column_name, data_type, is_nullable, column_default
FROM information_schema.columns
WHERE table_schema IN ('public')
  AND (
    table_name ILIKE '%actor%'
    OR table_name ILIKE '%organization%'
    OR table_name ILIKE '%space%'
    OR table_name ILIKE '%member%'
    OR table_name ILIKE '%role%'
    OR table_name ILIKE '%user%'
    OR table_name ILIKE '%app_user%'
    OR column_name ILIKE '%actor%'
    OR column_name ILIKE '%organization%'
    OR column_name ILIKE '%space%'
    OR column_name ILIKE '%member%'
    OR column_name ILIKE '%role%'
    OR column_name ILIKE '%status%'
    OR column_name ILIKE '%user%'
    OR column_name ILIKE '%app_user%'
    OR column_name ILIKE '%auth0%'
  )
ORDER BY table_schema, table_name, ordinal_position;

Recommended metadata query group C: primary keys, unique constraints and foreign keys
SELECT
  tc.table_schema,
  tc.table_name,
  tc.constraint_name,
  tc.constraint_type,
  kcu.column_name,
  ccu.table_schema AS foreign_table_schema,
  ccu.table_name AS foreign_table_name,
  ccu.column_name AS foreign_column_name
FROM information_schema.table_constraints tc
LEFT JOIN information_schema.key_column_usage kcu
  ON tc.constraint_name = kcu.constraint_name
  AND tc.table_schema = kcu.table_schema
LEFT JOIN information_schema.constraint_column_usage ccu
  ON tc.constraint_name = ccu.constraint_name
  AND tc.table_schema = ccu.table_schema
WHERE tc.table_schema IN ('public')
  AND (
    tc.table_name ILIKE '%actor%'
    OR tc.table_name ILIKE '%organization%'
    OR tc.table_name ILIKE '%space%'
    OR tc.table_name ILIKE '%member%'
    OR tc.table_name ILIKE '%role%'
    OR tc.table_name ILIKE '%user%'
    OR tc.table_name ILIKE '%app_user%'
  )
ORDER BY tc.table_schema, tc.table_name, tc.constraint_type, tc.constraint_name, kcu.ordinal_position;

Recommended metadata query group D: indexes
SELECT
  schemaname,
  tablename,
  indexname,
  indexdef
FROM pg_indexes
WHERE schemaname IN ('public')
  AND (
    tablename ILIKE '%actor%'
    OR tablename ILIKE '%organization%'
    OR tablename ILIKE '%space%'
    OR tablename ILIKE '%member%'
    OR tablename ILIKE '%role%'
    OR tablename ILIKE '%user%'
    OR tablename ILIKE '%app_user%'
  )
ORDER BY schemaname, tablename, indexname;

Recommended metadata query group E: RLS status
SELECT
  schemaname,
  tablename,
  rowsecurity
FROM pg_tables
WHERE schemaname IN ('public')
  AND (
    tablename ILIKE '%actor%'
    OR tablename ILIKE '%organization%'
    OR tablename ILIKE '%space%'
    OR tablename ILIKE '%member%'
    OR tablename ILIKE '%role%'
    OR tablename ILIKE '%user%'
    OR tablename ILIKE '%app_user%'
  )
ORDER BY schemaname, tablename;

Recommended metadata query group F: RLS policies
SELECT
  schemaname,
  tablename,
  policyname,
  permissive,
  roles,
  cmd,
  qual,
  with_check
FROM pg_policies
WHERE schemaname IN ('public')
  AND (
    tablename ILIKE '%actor%'
    OR tablename ILIKE '%organization%'
    OR tablename ILIKE '%space%'
    OR tablename ILIKE '%member%'
    OR tablename ILIKE '%role%'
    OR tablename ILIKE '%user%'
    OR tablename ILIKE '%app_user%'
  )
ORDER BY schemaname, tablename, policyname;
